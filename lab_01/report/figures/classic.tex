Схема классического нерекурсивного бинарного поиска представлена на рисунке~\ref{fig:classic}.

\begin{figure}[H]
\centering
\begin{tikzpicture}[flow]
\node[term] (start) at (0,0.0) {Начало: $a,N,x$};
\node[proc] (init) at (0,-1.6) {$L\gets0$, $R\gets N$};
\node[decision] (loop) at (0,-3.9) {$L<R$};
\node[proc] (mid) at (0,-6.2) {$M\gets L+$\\$\lfloor(R-L)/2\rfloor$};
\node[decision] (lt) at (0,-8.5) {$a_M<x$};
\node[decision] (gt) at (0,-11.3) {$a_M>x$};
\node[proc] (left) at (5.2,-8.5) {$L\gets M+1$};
\node[proc] (right) at (-5.2,-11.3) {$R\gets M$};
\node[term] (found) at (0,-13.6) {Возврат $M$};
\node[term] (miss) at (-5.2,-3.9) {Возврат $-1$};
\draw[arrow] (start)--(init);
\draw[arrow] (init)--(loop);
\draw[arrow] (loop)--node[left]{Да}(mid);
\draw[arrow] (loop)--node[above]{Нет}(miss);
\draw[arrow] (mid)--(lt);
\draw[arrow] (lt)--node[left]{Нет}(gt);
\draw[arrow] (lt)--node[above]{Да}(left);
\draw[arrow] (gt)--node[left]{Нет}(found);
\draw[arrow] (gt)--node[above]{Да}(right);
\draw[arrow] (left.east)--(7.5,-8.5)--(7.5,-2.5)--(0,-2.5)--(loop.north);
\draw[arrow] (right.west)--(-7.5,-11.3)--(-7.5,-2.5)--(0,-2.5)--(loop.north);
\end{tikzpicture}
\caption{Схема классического нерекурсивного бинарного поиска}\label{fig:classic}
\end{figure}
